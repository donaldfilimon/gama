private func oracleWrite(_ name: String, bytes: [UInt8]) throws {
    guard let output = ProcessInfo.processInfo.environment["GAMA_PARITY_OUTPUT"] else { return }
    let url = URL(fileURLWithPath: output).appendingPathComponent(name)
    try FileManager.default.createDirectory(at: url.deletingLastPathComponent(), withIntermediateDirectories: true)
    try Data(bytes).write(to: url, options: .atomic)
}
private func oracleJSON(_ name: String, _ value: Any) throws {
    let data = try JSONSerialization.data(withJSONObject: value, options: [.prettyPrinted, .sortedKeys, .withoutEscapingSlashes])
    try oracleWrite(name, bytes: Array(data) + [10])
}
private func quad(_ r: Rect) -> [Int] { [r.origin.x, r.origin.y, r.size.width, r.size.height] }
private func selection(_ s: Selection) -> [Int] { [s.anchor, s.head] }
private func cells(_ b: CellBuffer) -> [[String: Any]] {
    (0..<b.size.height).flatMap { y in (0..<b.size.width).map { x in
        let c = b.cell(atX: x, y: y)!
        return ["x": x, "y": y, "text": String(c.character), "continuation": c.isContinuation, "sgr": Int(c.style.attributes.rawValue)]
    } }
}
private func tree(_ n: LaidOutNode) -> [String: Any] {
    var value: [String: Any] = ["frame": quad(n.frame), "children": n.children.map(tree)]
    switch n.node {
    case .text(let s, _): value["kind"] = "text"; value["text"] = s
    case .interactive(let id, let focus, _): value["kind"] = "interactive"; value["id"] = String(id.raw); value["focusable"] = focus
    default: value["kind"] = String(describing: n.node).split(separator: "(").first.map(String.init) ?? "unknown"
    }
    return value
}
private func grid(_ laid: LaidOutNode, _ size: Size) -> CellBuffer {
    var b = CellBuffer(size: size); CellPainter.paint(laid, into: &b); return b
}
@Component private struct OracleCounter {
    let label: String
    @Reactive var count: Int = 0
    var body: some View { Button("\(label) \(count)") { count += 1 } }
}
private final class OracleModel { var visible = true; var items = [7, 3] }
private struct OracleStateApp: App {
    let model: OracleModel; let identified: Bool
    init() { model = OracleModel(); identified = false }
    init(model: OracleModel, identified: Bool) { self.model = model; self.identified = identified }
    var scenes: some Scene {
        Window("State", id: "main", role: .primary) {
            if model.visible {
                VStack(spacing: 0) {
                    if identified { IdentifiedForEach(model.items, id: { NodeID(raw: UInt64($0)) }) { OracleCounter(label: "i\($0)") } }
                    else { ForEach(model.items) { OracleCounter(label: "i\($0)") } }
                }
            } else { Text("off") }
        }
    }
}
private struct OracleActionApp: App {
    let text = Signal(""); let count = Signal(0); let enabled = Signal(true)
    var scenes: some Scene {
        Window("Actions", id: "main", role: .primary) {
            VStack(spacing: 0) {
                TextField("Name", text: text.binding())
                Button("Add") { count.update { $0 += 1 } }.actionIdentity(ActionID("add"), shortcut: .ctrl("a")).disabled(!enabled.get())
            }
        }
    }
}
private final class OracleEvents: @unchecked Sendable { var values: [String] = [] }
private struct OraclePlugin: GamaPluginProtocol {
    let manifest: PluginManifest; let events: OracleEvents
    init(_ id: PluginID, _ events: OracleEvents, requires: [Capability] = []) {
        manifest = PluginManifest(id: id, version: PluginVersion(major: 1, minor: 0, patch: 0), requires: requires); self.events = events
    }
    mutating func activate(in context: PluginContext) throws(PluginError) { events.values.append("activate:" + manifest.id.raw) }
    mutating func deactivate() { events.values.append("deactivate:" + manifest.id.raw) }
    func commands() -> [PluginCommand] {
        let events = events
        return [PluginCommand(id: "run", title: "Run") { context in events.values.append("run:" + context.plugin.raw) }]
    }
    func render(slot: SlotID, in context: BuildContext) -> RenderNode { .interactive(id: context.id, focusable: true, child: .text(manifest.id.raw, style: .plain)) }
}
